\documentclass[border=2pt]{standalone}
\usepackage[dvipsnames,svgnames,x11names]{xcolor}

\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usetikzlibrary{calc,arrows.meta,positioning,intersections,fit,shapes.geometric}
\usepgfplotslibrary{fillbetween}
\usetikzlibrary{spy}
\newcommand{\funfigcoordx}[2]{%
  \pgfplotspointgetcoordinates{(#1)}%
  \ensuremath{\pgfmathprintnumber[fixed,precision=#2]{\pgfkeysvalueof{/data point/x}}}%
}
\newcommand{\funfigcoordy}[2]{%
  \pgfplotspointgetcoordinates{(#1)}%
  \ensuremath{\pgfmathprintnumber[fixed,precision=#2]{\pgfkeysvalueof{/data point/y}}}%
}
\newcommand{\funfigcoordxy}[2]{%
  \pgfplotspointgetcoordinates{(#1)}%
  \ensuremath{(\pgfmathprintnumber[fixed,precision=#2]{\pgfkeysvalueof{/data point/x}},\pgfmathprintnumber[fixed,precision=#2]{\pgfkeysvalueof{/data point/y}})}%
}

\begin{tikzpicture}[spy using outlines={rectangle,magnification=4,size=2cm}]
\begin{axis}[
  width=8.6cm,

  axis line shift=6.5pt,
  grid=major,
  grid style={gray!25,line width=0.2pt},
  xlabel={$x$},
  xmin=-1.4,
  xmax=1.4,
  ylabel={$y$},
  ymin=-1.4,
  ymax=1.4,
  legend pos=north east,
  legend style={draw=none,fill=none,cells={anchor=west}},
]
\addplot+ gnuplot[raw gnuplot,empty line=jump,MidnightBlue,line width=1pt,mark=none,name path=circle-path] {
set xrange [-1.4:1.4];
set yrange [-1.4:1.4];
set contour;
unset surface;
set view map;
set cntrparam levels discrete 0;
set isosamples 100;
set samples 100;
splot (x**2 + y**2)-(1);
} coordinate[pos=0.12] (P);
\addlegendentry{$x^2+y^2=1$}
\addplot+ [Maroon,dashed,line width=0.8pt,mark=none,name path=diagonal-path,domain=-1.2:1.2,samples=80] {x} node[pos=0.72,sloped,font=\scriptsize] {$y=x$};
\addlegendentry{$y=x$}
\draw[fill=Dandelion,draw=black!70,line width=0.45pt] (P) circle (0.75mm);
\node[anchor=south west,font=\scriptsize] at ([shift={(6pt,8pt)}]P) {$P=(\funfigcoordx{P}{2},\funfigcoordy{P}{2})$};
\path[name intersections={of=circle-path and diagonal-path,by={I1,I2}}];
\draw[fill=OrangeRed,draw=black!70,line width=0.45pt] (I1) circle (0.7mm);

\node[anchor=south west,font=\scriptsize] at ([shift={(-74pt,10pt)}]I2) {$I_2=(\funfigcoordx{I2}{3},\funfigcoordy{I2}{3})$};

\coordinate (funfigSpyIn6) at (axis cs:0.82,-0.82);
\spy[circle,magnification=3.2,size=1.4cm,connect spies,black!55,line width=0.45pt] on (funfigSpyOn6) in node at (funfigSpyIn6);
\end{axis}
\end{tikzpicture}
